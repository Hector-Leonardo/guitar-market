rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Usuarios: Solo pueden leer/escribir su propio documento
    match /users/{uid} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.auth.uid == uid;
    }

    // Vendedores: Todos pueden leer, solo el vendedor puede escribir el suyo
    match /sellers/{uid} {
      allow read: if true;
      allow create, update: if request.auth != null && request.auth.uid == uid;
    }

    // Productos: Visible si es público O si el usuario es el vendedor
    match /products/{productId} {
      allow read: if resource.data.visible == true ||
        (request.auth != null && request.auth.uid == resource.data.sellerId);
      allow create: if request.auth != null &&
        request.auth.uid == request.resource.data.sellerId;
      allow update, delete: if request.auth != null &&
        request.auth.uid == resource.data.sellerId;
    }

    // Órdenes: El comprador y vendedor pueden leer. 
    // Se permite lectura con filtros (para getSellerStats, etc.)
    match /orders/{orderId} {
      allow read: if request.auth != null &&
        (request.auth.uid == resource.data.buyerId ||
         request.auth.uid == resource.data.sellerId);
      allow create: if request.auth != null &&
        request.auth.uid == request.resource.data.buyerId;
      allow update: if request.auth != null &&
        (request.auth.uid == resource.data.buyerId ||
         request.auth.uid == resource.data.sellerId);
    }

    // Feedback: Todos pueden leer, solo autenticados pueden crear
    match /feedback/{feedbackId} {
      allow read: if true;
      allow create: if request.auth != null &&
        request.auth.uid == request.resource.data.buyerId;
    }

    // Envíos: Solo el propietario puede leer/escribir
    match /shipments/{shipmentId} {
      allow read: if request.auth != null &&
        request.auth.uid == resource.data.userId;
      allow create: if request.auth != null &&
        request.auth.uid == request.resource.data.userId;
      allow update, delete: if request.auth != null &&
        request.auth.uid == resource.data.userId;
    }
  }
}
