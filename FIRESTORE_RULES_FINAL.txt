rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ============== USUARIOS ==============
    match /users/{uid} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.auth.uid == uid;
    }

    // ============== VENDEDORES ==============
    match /sellers/{uid} {
      allow read: if true;
      allow create, update: if request.auth != null && request.auth.uid == uid;
    }

    // ============== PRODUCTOS ==============
    match /products/{productId} {
      allow read: if resource.data.visible == true || (request.auth != null && request.auth.uid == resource.data.sellerId);
      allow create: if request.auth != null && request.auth.uid == request.resource.data.sellerId;
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.sellerId;
    }

    // ============== ÓRDENES ==============
    // Cualquier usuario autenticado puede LEER órdenes (Para getSellerStats)
    // Solo comprador y vendedor pueden ver sus órdenes específicas (filtradas por query)
    match /orders/{orderId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.buyerId;
      allow update: if request.auth != null && (request.auth.uid == resource.data.buyerId || request.auth.uid == resource.data.sellerId);
    }

    // ============== FEEDBACK ==============
    match /feedback/{feedbackId} {
      allow read: if true;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.buyerId;
    }

    // ============== ENVÍOS ==============
    match /shipments/{shipmentId} {
      allow read: if request.auth != null && request.auth.uid == resource.data.userId;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.userId;
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.userId;
    }
  }
}
